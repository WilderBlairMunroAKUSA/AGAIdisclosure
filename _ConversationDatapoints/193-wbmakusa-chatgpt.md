### 🧱 Debian Bookworm i3 Minimal Install Script
```bash
#!/usr/bin/env bash
# ────────────────────────────────────────────────
# wpbme-debian-i3-bootstrap.sh
# A minimal, reproducible setup for Wilder + Aurora
# ────────────────────────────────────────────────

set -e

echo "[*] Updating base system..."
sudo apt update && sudo apt full-upgrade -y

echo "[*] Installing core packages..."
sudo apt install -y \
  i3 i3status i3lock dmenu lightdm network-manager \
  alacritty thunar feh picom pavucontrol \
  neovim tmux git curl wget htop unzip \
  fonts-firacode fonts-noto-color-emoji \
  python3 python3-pip python3-venv

echo "[*] Enabling services..."
sudo systemctl enable lightdm
sudo systemctl enable NetworkManager

echo "[*] Creating user config directories..."
mkdir -p ~/.config/{i3,alacritty,tmux}
mkdir -p ~/Projects ~/Downloads ~/Pictures ~/Documents

echo "[*] Writing minimal i3 config..."
cat <<'EOF' > ~/.config/i3/config
set $mod Mod4
font pango:Fira Code 10

# launch terminal / browser / lock
bindsym $mod+Return exec alacritty
bindsym $mod+b exec firefox
bindsym $mod+Shift+e exec i3-msg exit

# movement
bindsym $mod+h focus left
bindsym $mod+j focus down
bindsym $mod+k focus up
bindsym $mod+l focus right

# resize
bindsym $mod+Shift+h resize shrink width 10 px or 10 ppt
bindsym $mod+Shift+l resize grow width 10 px or 10 ppt

# restart / reload
bindsym $mod+Shift+r restart
bindsym $mod+Shift+c reload

exec --no-startup-id picom --experimental-backends
exec --no-startup-id feh --bg-scale ~/Pictures/bg.jpg
exec --no-startup-id nm-applet
exec --no-startup-id pavucontrol
EOF

echo "[*] Writing minimal Alacritty config..."
cat <<'EOF' > ~/.config/alacritty/alacritty.yml
font:
  normal:
    family: "Fira Code"
    style: Regular
  size: 10.0
window:
  padding:
    x: 4
    y: 4
colors:
  primary:
    background: '0x0a0a0a'
    foreground: '0xe0e0e0'
EOF

echo "[*] Setup complete!"
echo "Reboot, choose 'i3' at login, and welcome to peacebrain mode."
```
